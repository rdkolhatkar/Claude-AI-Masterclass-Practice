---------------------------------------------------------------------
1. CREATE THE SCHEMA
---------------------------------------------------------------------

CREATE SCHEMA IF NOT EXISTS MultiAgentAutomationWorkflow;

-- Select the schema so that all following statements run inside it
USE MultiAgentAutomationWorkflow;


---------------------------------------------------------------------
2. CREATE TABLE: Customers
---------------------------------------------------------------------
-- Fields: unique_Id, name, city, age

CREATE TABLE IF NOT EXISTS Customers (
    unique_Id   INT           NOT NULL AUTO_INCREMENT,
    name        VARCHAR(100)  NOT NULL,
    city        VARCHAR(100)  NOT NULL,
    age         INT           NOT NULL,
    CONSTRAINT pk_customers PRIMARY KEY (unique_Id),
    CONSTRAINT chk_customers_age CHECK (age > 0 AND age <= 150)
);


---------------------------------------------------------------------
3. CREATE TABLE: Orders
---------------------------------------------------------------------
-- Fields: order_Id, customer_Id, subject, company, order_date
-- Foreign Key: customer_Id REFERENCES Customers(unique_Id)

CREATE TABLE IF NOT EXISTS Orders (
    order_Id     INT           NOT NULL AUTO_INCREMENT,
    customer_Id  INT           NOT NULL,
    subject      VARCHAR(200)  NOT NULL,
    company      VARCHAR(150)  NOT NULL,
    order_date   DATE          NOT NULL,
    CONSTRAINT pk_orders PRIMARY KEY (order_Id),
    CONSTRAINT fk_orders_customer
        FOREIGN KEY (customer_Id)
        REFERENCES Customers (unique_Id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

-- Note: ON DELETE CASCADE means deleting a customer also deletes that
-- customer's orders. Remove it if you prefer to block such deletes instead.

5. DELETE A SPECIFIC ROW FROM EACH TABLE
---------------------------------------------------------------------

-- 5a. Delete a specific row from Orders (order_Id = 3)

WHERE order_Id = 3;

-- 5b. Delete a specific row from Customers (unique_Id = 5)

WHERE unique_Id = 5;

---------------------------------------------------------------------
6. UPDATE / MODIFY A SPECIFIC VALUE OR ROW IN EACH TABLE
---------------------------------------------------------------------

-- 6a. Update the city of the customer whose unique_Id = 2
UPDATE Customers
SET city = 'Chennai'
WHERE unique_Id = 2;

-- 6b. Update the subject and company of the order whose order_Id = 1
UPDATE Orders
SET subject = 'Laptop Purchase - Upgraded',
    company = 'Lenovo'
WHERE order_Id = 1;

-- Verify

SELECT * FROM Orders    WHERE order_Id  = 1;

7. DELETE ALL RECORDS FROM EACH TABLE
---------------------------------------------------------------------
-- Orders depends on Customers through a foreign key, so delete the
-- child table (Orders) first, then the parent table (Customers).

-- 7a. Delete all rows from Orders (child table)

-- 7b. Delete all rows from Customers (parent table)

8. ALTER / MODIFY CONSTRAINTS ON EACH TABLE
---------------------------------------------------------------------

-- 8a. Customers: add a UNIQUE constraint on name + city combination
ALTER TABLE Customers
    ADD CONSTRAINT uq_customers_name_city UNIQUE (name, city);

-- 8b. Customers: change the age CHECK constraint (drop old, add new)
ALTER TABLE Customers
    DROP CHECK chk_customers_age;

ALTER TABLE Customers
    ADD CONSTRAINT chk_customers_age CHECK (age >= 18 AND age <= 100);

ALTER TABLE Customers
    ALTER COLUMN city SET DEFAULT 'Unknown';

-- 8d. Orders: change the foreign key behaviour
--     (drop the existing FK first, then re-create it)
ALTER TABLE Orders
    DROP FOREIGN KEY fk_orders_customer;

ALTER TABLE Orders
    ADD CONSTRAINT fk_orders_customer
        FOREIGN KEY (customer_Id)
        REFERENCES Customers (unique_Id)

-- 8e. Orders: make order_date default to today's date
ALTER TABLE Orders
    ALTER COLUMN order_date SET DEFAULT (CURRENT_DATE);

-- 8f. Orders: add an index on customer_Id to speed up joins
CREATE INDEX idx_orders_customer_id ON Orders (customer_Id);
